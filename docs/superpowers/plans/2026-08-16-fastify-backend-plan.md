# Fastify Backend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a testable Fastify ESM server that securely serves the existing frontend, exposes a health endpoint, and shuts down cleanly without changing frontend tooling.

**Architecture:** `server/app.js` creates a configured Fastify instance with CORS, static serving, logger options, and sanitized JSON error handlers. Focused route, controller, and service modules provide `GET /api/health`; `server/server.js` owns environment-based listening and signal handling.

**Tech Stack:** Node.js ESM, Fastify, `@fastify/cors`, `@fastify/static`, Node native test runner, existing esbuild and Tailwind v4 commands.

**Spec:** `docs/superpowers/specs/2026-08-16-fastify-backend-design.md`

## Global Constraints

- Preserve the existing `build:css`, `dev:css`, `build:js`, `build`, and `test` script values.
- Keep the repository ESM configuration (`"type": "module"`).
- Keep `npm run test` on the native Node test runner and add server tests through `tests/*.test.mjs`.
- Serve only root-level HTML, `js/bundle.js`, `css/output.css`, and optional `images`, `icons`, and `fonts` asset directories.
- Reject literal or encoded traversal, null bytes, hidden path segments, `.env`, `.git`, `package.json`, tests, source modules, and other non-frontend files.
- Use `logger: false` for application-factory tests and enable Fastify’s built-in logger for the process entrypoint.
- Return sanitized JSON for errors and 404s, with `Internal Server Error` for unknown server failures and no stack traces in response bodies.
- Load `.env` only in `dev:server` through `node --env-file-if-exists=.env`.
- Await `app.close()` for both `SIGTERM` and `SIGINT`.

- [ ] **Step 3: Implement the static allowlist**

Use the repository root derived from the module location, not the caller’s working directory. The `allowedPath` predicate must decode the URL, reject malformed encoding, null bytes, backslashes, `.`/`..` segments, and hidden segments, then allow only root HTML, the two built files, or files below `images`, `icons`, and `fonts`:

```js
// server/plugins/static.js
import fastifyStatic from '@fastify/static';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

export const PROJECT_ROOT = fileURLToPath(new URL('../..', import.meta.url));

const OPTIONAL_ASSET_DIRECTORIES = new Set(['images', 'icons', 'fonts']);

function isAllowedPath(pathName) {
  let decodedPath;
  try {
    decodedPath = decodeURIComponent(pathName);
  } catch {
    return false;
  }

  if (decodedPath.includes('\0') || decodedPath.includes('\\')) return false;

  const relativePath = decodedPath.replace(/^\/+/, '');
  const segments = relativePath.split('/');
  if (segments.some(segment => !segment || segment === '.' || segment === '..' || segment.startsWith('.'))) {
    return false;
  }

  if (/^[^/]+\.html$/i.test(relativePath)) return true;
  if (relativePath === 'js/bundle.js' || relativePath === 'css/output.css') return true;
  return segments.length > 1 && OPTIONAL_ASSET_DIRECTORIES.has(segments[0]);
}

export async function staticPlugin(app, { root = PROJECT_ROOT } = {}) {
  await app.register(fastifyStatic, {
    root: path.resolve(root),
    index: false,
    allowedPath: isAllowedPath
  });

  app.get('/', (_request, reply) => reply.sendFile('index.html'));
}
```

`@fastify/static` remains responsible for its normal path resolution while `allowedPath` prevents requests from reaching it for disallowed project files. The explicit root route sends only the known frontend entrypoint.

- [ ] **Step 4: Implement the app factory and sanitized JSON handlers**

Create the factory with a configurable logger and a stable error policy:

```js
// server/app.js
import Fastify from 'fastify';
import { corsPlugin } from './plugins/cors.js';
import { staticPlugin, PROJECT_ROOT } from './plugins/static.js';
import { healthRoutes } from './routes/health.routes.js';

function errorBody(code, message) {
  return { error: { code, message } };
}

function safeStatusCode(error) {
  return Number.isInteger(error.statusCode) && error.statusCode >= 400 && error.statusCode < 500
    ? error.statusCode
    : 500;
}

export function buildApp({ logger = true, staticRoot = PROJECT_ROOT } = {}) {
  const app = Fastify({ logger });

  app.setErrorHandler((error, request, reply) => {
    const statusCode = safeStatusCode(error);
    const isServerError = statusCode >= 500;
    if (isServerError) request.log.error({ err: error }, 'Request failed');

    return reply.status(statusCode).send(errorBody(
      isServerError ? 'INTERNAL_SERVER_ERROR' : (error.code || 'REQUEST_ERROR'),
      isServerError ? 'Internal Server Error' : error.message
    ));
  });

  app.setNotFoundHandler((_request, reply) => {
    return reply.status(404).send(errorBody('NOT_FOUND', 'Route not found'));
  });

  app.register(corsPlugin);
  app.register(staticPlugin, { root: staticRoot });
  app.register(healthRoutes, { prefix: '/api' });
  return app;
}
```

- [ ] **Step 1: Add a child-process shutdown smoke test before the entrypoint**

Add these imports to the existing import block, then append this test and its
helpers to `tests/server.test.mjs`:

```js
import { spawn } from 'node:child_process';
import { once } from 'node:events';

async function waitForListening(child) {
  return new Promise((resolve, reject) => {
    let output = '';
    const timeout = setTimeout(() => reject(new Error(`server did not start: ${output}`)), 5000);
    const onData = (chunk) => {
      output += chunk.toString();
      if (/listening/i.test(output)) {
        clearTimeout(timeout);
        resolve();
      }
    };
    child.stdout.on('data', onData);
    child.stderr.on('data', onData);
    child.once('error', (error) => {
      clearTimeout(timeout);
      reject(error);
    });
    child.once('exit', (code, signal) => {
      if (code !== null || signal !== null) {
        clearTimeout(timeout);
        reject(new Error(`server exited before listening: ${code ?? signal}`));
      }
    });
  });
}

test('server exits cleanly after SIGTERM', async (t) => {
  const child = spawn(process.execPath, ['server/server.js'], {
    cwd: process.cwd(),
    env: { ...process.env, HOST: '127.0.0.1', PORT: '0' }
  });

  t.after(() => {
    if (child.exitCode === null) child.kill('SIGKILL');
  });

  await waitForListening(child);
  child.kill('SIGTERM');
  const [code, signal] = await once(child, 'exit');
  assert.equal(code, 0);
  assert.equal(signal, null);
});
```

- [ ] **Step 3: Implement the listener and signal handling**

Implement `startServer` with defaults parsed from the environment. The signal handler must reuse one close promise, await `app.close()`, and avoid forced process termination:

```js
import { pathToFileURL } from 'node:url';
import { buildApp } from './app.js';

const defaultPort = Number.parseInt(process.env.PORT ?? '3000', 10);
const defaultHost = process.env.HOST ?? '0.0.0.0';

export async function startServer({ port = defaultPort, host = defaultHost } = {}) {
  const app = buildApp({ logger: true });
  let closePromise;

  const shutdown = async (signal) => {
    closePromise ??= app.close();
    app.log.info({ signal }, 'Server shut down');
    await closePromise;
  };

  process.once('SIGTERM', shutdown);
  process.once('SIGINT', shutdown);

  try {
    await app.listen({ port, host });
    return app;
  } catch (error) {
    await app.close();
    throw error;
  }
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  startServer().catch((error) => {
    console.error(error);
    process.exitCode = 1;
  });
}
```
